// ---- first-failure classifier: WHY a real command misses the template ----
function firstFailure(tool, cmd) {
  if (TEMPLATE.test(cmd)) return null;
  if (/[\r\n]/.test(cmd)) return 'multiline (newline or backtick continuation)';
  if (/^\s*\$\w+\s*=/.test(cmd)) return 'assignment prefix ($x = curl ...)';
  if (/(?:^|[^|])\|(?!\|)/.test(cmd)) return 'piped (| ...)';
  if (/(?:^|\s)>>?\s*\S/.test(cmd)) return 'output redirect (> file)';
  if (/(?:;|&&|\|\|)/.test(cmd)) return 'chained (; && ||)';
  if (/\b(?:Invoke-RestMethod|Invoke-WebRequest|irm|iwr)\b/i.test(cmd)) return 'PowerShell-native client (not admissible)';
  if (/\b(?:node|python|py)\b\s+-(?:e|c)\b/.test(cmd)) return 'inline interpreter';
  if (!/^\s*curl(\.exe)?\b/.test(cmd)) return 'does not start with the client token';
  if (/^\s*curl\b(?!\.exe)/.test(cmd) && tool === 'PowerShell') return 'bare `curl` in PowerShell (alias for Invoke-WebRequest)';
  if (!/(?:^|\s)-q(?:\s|$)/.test(cmd)) return 'no -q (ambient .curlrc not disabled)';
  const urlm = /api\.notion\.com\/v1\/[^\s"'`]*/.exec(cmd);
  if (urlm && /[$`]/.test(urlm[0])) return 'URL interpolated (variable in the URL, target unreadable)';
  if (urlm && !new RegExp('^api\\.notion\\.com/v1/(?:pages|blocks|databases)/(?:' + ID + ')(?:/children)?$').test(urlm[0].replace(/^https?:\/\//, '').replace(/\?.*$/, '')))
    return 'URL not one of the permitted endpoint shapes';
  const bodym = /--data-binary\s+"@([^"]*)"/.exec(cmd);
  if (bodym) {
    if (/[$`]/.test(bodym[1])) return 'body path interpolated (@$var)';
    if (!new RegExp('^(?:[A-Za-z]:[\\\\/]|/)').test(bodym[1])) return 'body path not absolute';
  }
  if (/(?:^|\s)(--data-raw|--data-ascii|--data-urlencode|--json|-d|--data)(?:\s|=)/.test(cmd)) return 'body flag other than --data-binary';
  if (/(?:^|\s)(-K|--config|-F|--form|-T|--upload-file)\b/.test(cmd)) return 'file/config body flag not in the template';
  return 'other token(s) outside the template';
}
